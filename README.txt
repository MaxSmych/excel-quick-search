━━━ ОПИСАНИЕ ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Live-поиск по любому Excel-файлу с выбором отображаемых колонок.
Стек: Python 3 / tkinter / tksheet / pandas / openpyxl


━━━ ЗАВИСИМОСТИ ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Требуется Python 3.8+ с pip.

Установка Python: https://www.python.org/downloads/
  ⚠️ Add Python to PATH — обязательная галочка при установке.

Проверка:
  python --version

Пакеты (pandas, openpyxl, tksheet) устанавливаются автоматически
при первом запуске. Нужен интернет.


━━━ ЗАПУСК ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Открыть поиск.bat       — двойной клик
  python excel_search.py  — из терминала


━━━ ИСПОЛЬЗОВАНИЕ ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  - Обзор → выбрать .xlsx / .xls / .xlsb
  - Чекбоксы слева — управление видимостью колонок
  - Колонка ✓ (первая в таблице) — отметка «найдено/обработано» для строки;
    сохраняется между запусками (привязка к содержимому строки, не к позиции)
  - Поиск — live, по всем выбранным колонкам одновременно
  - Пофильтры у колонок (поля слева) — сохраняются между запусками
  - Клик по заголовку колонки — сортировка (повторный клик — обратный порядок)
  - Drag заголовка колонки — изменение порядка
  - Авто↺ — polling файла каждые 5 сек, перезагрузка при изменении
  - Экспорт — выгрузка текущей выборки в Excel (.xlsx) или CSV (UTF-8 BOM)
  - Настройки видимости колонок сохраняются в excel_search_settings.json


━━━ ФАЙЛЫ ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

                               не требует передачи, можно удалить
